# 1. Update system and install required tools
sudo apt update -y && sudo apt install -y git maven openjdk-17-jdk docker.io
sudo systemctl start docker

# 2. Configure Java 17 path (Prevents the compiler error)
export JAVA_HOME=/usr/lib/jvm/java-17-openjdk-amd64
export PATH=$JAVA_HOME/bin:$PATH

# 3. Clone the WAR repository and enter folder
git clone https://github.com/anujyog1/carrepo.git
cd carrepo

# 4. Package the project into a .war file
mvn clean package

# 5. Create the Tomcat Dockerfile
cat << 'EOF' > Dockerfile
FROM tomcat:9.0-jdk8
RUN rm -rf /usr/local/tomcat/webapps/*
COPY target/*.war /usr/local/tomcat/webapps/ROOT.war
EXPOSE 8080
CMD ["catalina.sh", "run"]
EOF

# 6. Build the Docker image
sudo docker build -t car-web-app .

# 7. Run the container on port 8080
sudo docker run -d --name car-web -p 8080:8080 car-web-app

# 8. Verify the container is running
sudo docker psc
